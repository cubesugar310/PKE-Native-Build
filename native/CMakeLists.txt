cmake_minimum_required(VERSION 3.24)
project(PKENativePrototype VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PKE_NATIVE_OFFLINE_CORE "Build deterministic core/tests without fetching GUI/audio dependencies" OFF)

if(NOT PKE_NATIVE_OFFLINE_CORE)
  include(cmake/Dependencies.cmake)
endif()

add_library(pke_native_core STATIC src/Config.cpp src/AssetLocator.cpp src/ChartParser.cpp src/RhythmEngine.cpp src/CharacterLoader.cpp src/CharacterAnimator.cpp src/AudioEngine.cpp src/Diagnostics.cpp src/Renderer.cpp src/GameSession.cpp src/LegacySaveImporter.cpp src/App.cpp)
target_include_directories(pke_native_core PUBLIC include)
if(NOT PKE_NATIVE_OFFLINE_CORE)
  target_link_libraries(pke_native_core PUBLIC nlohmann_json::nlohmann_json tinyxml2::tinyxml2 SDL2::SDL2 SDL2_image::SDL2_image SDL2_ttf::SDL2_ttf)
  target_compile_definitions(pke_native_core PUBLIC PKE_NATIVE_USE_NLOHMANN=1 PKE_NATIVE_WITH_AUDIO_DEPS=1 PKE_NATIVE_WITH_SDL=1)
  target_include_directories(pke_native_core PUBLIC ${miniaudio_SOURCE_DIR} ${stb_SOURCE_DIR})
endif()

add_executable(pke_native src/main.cpp)
target_link_libraries(pke_native PRIVATE pke_native_core)
if(NOT PKE_NATIVE_OFFLINE_CORE)
  target_link_libraries(pke_native PRIVATE SDL2::SDL2 SDL2::SDL2main)
  target_compile_definitions(pke_native PRIVATE PKE_NATIVE_WITH_SDL=1)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(pke_native_tests tests/test_main.cpp tests/test_config.cpp tests/test_asset_locator.cpp tests/test_chart_parser.cpp tests/test_chart_ejected_golden.cpp tests/test_rhythm_judgement.cpp tests/test_rhythm_sustain.cpp tests/test_character_loader.cpp tests/test_sparrow_geometry.cpp tests/test_character_animator.cpp tests/test_audio_mix_cursor.cpp tests/test_audio_state.cpp tests/test_diagnostics.cpp tests/test_render_math.cpp tests/test_game_session.cpp tests/test_song_lifecycle.cpp tests/test_acceptance_fixture.cpp tests/test_legacy_save_import.cpp)
  target_include_directories(pke_native_tests PRIVATE tests)
  target_compile_definitions(pke_native_tests PRIVATE PKE_TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  target_link_libraries(pke_native_tests PRIVATE pke_native_core)
  add_test(NAME config COMMAND pke_native_tests)
  add_test(NAME asset_locator COMMAND pke_native_tests)
  add_test(NAME chart_parser COMMAND pke_native_tests)
  add_test(NAME chart_ejected COMMAND pke_native_tests)
  add_test(NAME rhythm COMMAND pke_native_tests)
  add_test(NAME character COMMAND pke_native_tests)
  add_test(NAME sparrow COMMAND pke_native_tests)
  add_test(NAME character_animator COMMAND pke_native_tests)
  add_test(NAME audio COMMAND pke_native_tests)
  add_test(NAME diagnostics COMMAND pke_native_tests)
  add_test(NAME render_math COMMAND pke_native_tests)
  add_test(NAME game_session COMMAND pke_native_tests)
  add_test(NAME song_lifecycle COMMAND pke_native_tests)
  add_test(NAME acceptance_fixture COMMAND pke_native_tests)
endif()
