cmake_minimum_required(VERSION 3.24)
project(PKENativePrototype VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PKE_NATIVE_OFFLINE_CORE "Build deterministic core/tests without fetching GUI/audio dependencies" OFF)

if(NOT PKE_NATIVE_OFFLINE_CORE)
  include(cmake/Dependencies.cmake)
endif()

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/quickjs-ng/quickjs.h")
  message(FATAL_ERROR "QuickJS submodule missing: run git submodule update --init native/third_party/quickjs-ng")
endif()
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "No standalone QuickJS installation" FORCE)
set(QJS_BUILD_LIBC OFF CACHE BOOL "Native compatibility has no std/os modules" FORCE)
add_subdirectory(third_party/quickjs-ng quickjs EXCLUDE_FROM_ALL)

add_library(pke_native_core STATIC src/Config.cpp src/AssetLocator.cpp src/ChartParser.cpp src/RhythmEngine.cpp src/CharacterLoader.cpp src/CharacterAnimator.cpp src/AudioEngine.cpp src/Diagnostics.cpp src/Renderer.cpp src/GameSession.cpp src/LegacySaveImporter.cpp src/App.cpp src/QuickJsRuntime.cpp src/PopulationRuntime.cpp src/QuickJsPopulationBackend.cpp src/AtomicSaveFile.cpp src/SaveService.cpp)
target_sources(pke_native_core PRIVATE src/AssetService.cpp)
target_sources(pke_native_core PRIVATE src/ApplicationFlow.cpp)
target_sources(pke_native_core PRIVATE src/UiRenderer.cpp)
target_sources(pke_native_core PRIVATE src/ApplicationView.cpp)
target_sources(pke_native_core PRIVATE src/CharacterPresentation.cpp)
if(PKE_NATIVE_OFFLINE_CORE)
  target_sources(pke_native_core PRIVATE src/OfflineJson.cpp)
endif()
target_include_directories(pke_native_core PUBLIC include)
find_package(Threads REQUIRED)
target_link_libraries(pke_native_core PUBLIC qjs Threads::Threads)
if(WIN32)
  # Windows 10's Native ICU C APIs; these are OS libraries, not browser DLLs.
  target_link_libraries(pke_native_core PUBLIC icuin icuuc ole32)
else()
  find_package(ICU REQUIRED COMPONENTS i18n uc)
  target_link_libraries(pke_native_core PUBLIC ICU::i18n ICU::uc)
endif()
if(NOT PKE_NATIVE_OFFLINE_CORE)
  target_link_libraries(pke_native_core PUBLIC nlohmann_json::nlohmann_json tinyxml2::tinyxml2 SDL2::SDL2 SDL2_image::SDL2_image SDL2_ttf::SDL2_ttf)
  target_compile_definitions(pke_native_core PUBLIC PKE_NATIVE_USE_NLOHMANN=1 PKE_NATIVE_WITH_AUDIO_DEPS=1 PKE_NATIVE_WITH_SDL=1)
  target_include_directories(pke_native_core PUBLIC ${miniaudio_SOURCE_DIR} ${stb_SOURCE_DIR})
endif()

add_executable(pke_native src/main.cpp)
target_link_libraries(pke_native PRIVATE pke_native_core)
set(PKE_COMPAT_SCRIPTS cpu-timing.js title-system.js population-social-sessions.js boss-battle.js song-difficulty.js population-imports.js population.js chart-utils.js catalog-imports.js catalog.js practice-replay.js character-presentation.js)
foreach(script IN LISTS PKE_COMPAT_SCRIPTS)
  add_custom_command(TARGET pke_native POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:pke_native>/compat"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/compat/${script}" "$<TARGET_FILE_DIR:pke_native>/compat/${script}")
endforeach()
if(NOT PKE_NATIVE_OFFLINE_CORE)
  target_link_libraries(pke_native PRIVATE SDL2::SDL2 SDL2::SDL2main)
  target_compile_definitions(pke_native PRIVATE PKE_NATIVE_WITH_SDL=1)
endif()
